LoopStatement -> loop StatementList end loop

ExitStatement -> exit when Expression ;

## Medjukod

### Loop Statement
```
pocetak:
    IMC<statements[0]>
    IMC<statements[1]>
    ...
    IMC<statements[n-1]>
    Jump pocetak
kraj:
```

### Exit statement
```
IMC<exp>
Load_Mem R1, Result exp
JumpIfNotZero R1, kraj
```

## Kod

```java
class LoopStatement extends Statement {
    public ArrayList<Statement> statements;

    public LoopStatement(ArrayList<statement> list) {
        this.statements = list;
    }

    @Override
    public void translate( BufferedWriter out ) throws IOException {
        String pocetak = ASTNode.genLab();
        String kraj = ASTNode.genLab();

        out.write( pocetak + ":" + "\n" );
        for (int i = 0; i < statements.size(); i++) {
            Statement current = statement[i];

            if (current instanceof ExitStatement) {
                ((ExitStatement)current).setEndLabel(kraj);
            }
            statements[i].translate( out );
        }
        out.write("Jump ", kraj + "\n");

        out.write( kraj + ":" + "\n" );
    }
}

class ExitStatement extends Statement {
    public Expression exp;
    public String endLabel;

    public ExitStatement(Expression exp, String endLabel) {
        this.exp = exp;
        this.endLabel = endLabel;
    }

    public setEndLabel(String endLabel) {
        this.endLabel = endLabel;
    }

    @Override
    public void translate( BufferedWriter out ) {
        exp.translate( out );
        exp.genLoad( "R1", out );
        out.write( "JumpIfNotZero R1" + endLabel + "\n");
    }
}
```
