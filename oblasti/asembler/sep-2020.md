# Zadatak
Definisati IA-32 kod i izgled aktivacionog sloga za funkciju covid_safety. Za predstavljanje
podataka tipa int kao i memorijskih adresa koriste se 4 bajta. Pretpostaviti da se rezultat funkcije
call_guard (koji je tipa int) smešta na stek, a rezultat funkcije covid_safety vraća kroz registar
ECX.

```c
struct event{
    int type;
    struct event* next;
};

int covid_safety(struct event *e, int location, int v_type) {
    int violations = 0;

    if (e->type == v_type) {
        violations += call_guard(v_type, location);
    }

    if (e->next != 0)
        return violations + covid_safety(e->next, location, v_type);
}
```

# Resenje

## Izgled strukture event
```
+------------------+
| next             | +4
+------------------+
| type             | +0
+------------------+
```

## Aktivacioni slog

```
+------------------+
| v_type           | [EBP+16]
+------------------+
| location         | [EBP+12]
+------------------+
| e                | [EBP+8]
+------------------+
| EID(Pret. adresa)| [EBP+4]
+------------------+
| Staro EBP        | <- EBP
+------------------+
| violations       | <- ESP [EBP-4]
+------------------+
```

## Asembler

```x86asm

covid_safety:
    PUSH EBP
    MOV EBP, ESP

    SUB ESP, 4

    ;e->type
    MOV EBX, [EBP+8]
    MOV EAX, [EBX]
    ;v_type
    MOV EDX, [EBP+16]
    CMP EDX, EAX
    JNE lab1

    ;rezervisemo na stack-u jedno mesto za rezultat funkcije
    PUSH EAX
    ;stavljamo na stack argumente za call_guard
    MOV EAX, [EBP+12]
    PUSH EAX
    MOV EAX, [EBP+16]
    PUSH EAX
    
    CALL call_guard
    ; skinemo sa stack-a argumente nakon poziva funkcije
    ADD ESP, 8

    ; ucitamo rezultat funkcije
    POP EAX
    ; saberemo violatsion sa EAX i odmah ga smestimo u vilations
    ADD [EBP-4], EAX
    
    lab2:
        ;e->next
        MOV EBX, [EBP+8]
        MOV EAX, [EBX+4]

        CMP EAX, 0
        JE kraj

        ; pushujemo na stack argumente za poziv rekurzije
        MOV
        ;v_type
        MOV EAX, [EBP+16]
        PUSH EAX
        ;location
        MOV EAX, [EBP+12]
        PUSH EAX
        ;e->next
        MOV EBX, [EBP+8]
        MOV EAX, [EBX+4]
        PUSH EAX
        ;zovemo funkciju
        CALL covid_safety
        ;skidamo sa stack-a argumente
        ADD ESP, 12
        
        ;rezultata funkcije je u ECX, sabiramo ga sa viloations i automatski vracamo rezultat funkcije kroz ECX
        ;msm da moze ovako da se napise, oni su ovo razdvojili u dva koraka na cs-u
        ADD ECX, [EBP-4]
    
    kraj:
        MOV ESP, EBP
        POP EBP
        RET

```
