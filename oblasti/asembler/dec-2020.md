# Zadatak

Definisati IA-32 kod i izgled aktivacionog sloga za funkciju allocate. Za predstavljanje
podataka tipa int kao i memorijskih adresa koriste se 4 bajta. Pretpostaviti da se rezultat funkcije
allocate smešta na stek.

```c
struct server{
    int id;
    int mem_available;
    int alloc_container;
    struct server* next;
};

int allocate(int task_id, int task_mem, struct server* s) {
    if (task_mem < s->mem_available){
        s->mem_available = s->mem_available-task_mem;
        s->alloc_container=task_id;
        return s->id;
    }

    if(s->next!=0)
        return allocate(task_id,task_mem,s->next);
    else return 0;
}
```

---

# Resenje

## Izgled strukture **server**

```
+------------------+
| next             | +12
+------------------+
| alloc-container  | +8
+------------------+
| mem_available    | +4
+------------------+
| id               | +0
+------------------+
```

## Aktivacioni slog

```
+------------------------+
| return result          | [EBP+20]
+------------------------+
| s                      | [EBP+16]
+------------------------+
| task_mem               | [EBP+12]
+------------------------+
| task_id                | [EBP+8]
+------------------------+
| EID (Adr. povratka)    | [EBP+4]
+------------------------+
| Staro EBP              | <- EBP, ESP
+------------------------+
```

## Asembler
```x86asm

allocate:
    PUSH EBP
    MOV EBP, ESP

    ;ucitamo task_mem
    MOV EAX, [EBP+12]
    ;ucitamo s->mem_available
    MOV EBX, [EBP+16]
    MOV EDX, [EBX+4]

    CMP EAX, EDX
    JGE lab1

    THEN_LABELA:
        SUB EAX, EDX
        MOV [EBX+4], EAX

        ;ucitamo task_id
        MOV ECX, [EBP+8]
        ;s->alloc_container = task_id
        MOV [EBX+8], ECX

        ;return s->id
        MOV EAX, [EBX]
        MOV [EBP+20], EAX
        JMP kraj

    lab1:
        ;ucitamo s->next, s nam je i dalje ucitan u EBX
        MOV EAX, [EBX+12]

        CMP EAX, 0
        JE lab2

        ;then
        ;rezervisemo mem za rezultat
        PUSH EAX
        ;guramo argumente na stack
        ;s->next
        MOV EAX, [EBX+12]
        PUSH EAX
        ;task_mem
        MOV EAX, [EBP+12]
        PUSH EAX
        ;task_ID
        MOV EAX, [EBP+8]
        PUSH EAX
        
        CALL allocate

        ;skidamo args sa stack-a
        ADD ESP, 12

        ;uzimamo rezultat funkcije
        POP EAX
        ;stavljamo na dno stack-a, u rezulatat
        MOV [EBP+20], EAX
        JMP kraj

    lab2:
        ;return 0
        MOV [EBP+20], 0

    kraj:
        MOV ESP, EBP
        POP EBP
        RET

```
