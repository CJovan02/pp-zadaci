class ExpressionInfo {
    public Type tip;
    public Object value;

    public ExpressionInfo(Type tip, Object value) {
        this.tip = tip;
        this.value = value;
    }
}

class FunctionDefinition {
    public List<ExpressionInfo>? arguments;
    public string name;

    public FunctionDefinition(List<ExpressionInfo>? arguments, string name) {
        this.arguments = arguments;
        this.name = name;
    }
}

FunctionCall ::= ID:imeFunkcije LEVA_ZAGRADA Arguments:args DESNA_ZAGRADA
    {:
        FunctionDefinition functionDefinition = parser.symbolTable.getFunction(imeFunkcije)

        // 1) Funkcija mora da posoji u tabeli simbola
        if (functionDefinition == null) {
            print("Ne mozes da pozivas funkciju koja nije definisana");
        }

        // 2) broj argumenata mora da bude isti kao i u deklaraciji funkcije
        if (functionDefinition.arguments.length != args.length) {
            print("Broj argumenata u pozivu se ne poklapaju sa definicijom funkcije");
        }

        // 3) Tipovi parametara u pozivu treba da se slazu sa tipovima u deklaraciji funkcije
        for (int i = 0; i < args.length; i++) {
            var argPoziva = args[i];
            var argDefinicije = functionDefinition.arguments[i];

            if (argPoziva.tip.tkind != argDefinicije.type.tkind) {
                print("Tip u pozivu se razlikuje sa tipom u definiciji");
            }
        }

    :}
