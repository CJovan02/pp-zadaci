import java_cup.runtime.*

%%

%class Lexer
%cup
%line
%column

%eofval{
    return new Symbol(sym.EOF);
%eofval}

// Makroi
slovo = [a-zA-Z]
cifra = [0-9]

%%

[\t\r\n ]+      { ; }

"."             { return new Symbol(sym.TACKA); }
"["             { return new Symbol(sym.LEVA_ZAGRADA); }
"]"             { return new Symbol(sym.DESNA_ZAGRADA); }

{slovo}({slovo}|{cifra}|_)* { return new Symbol(sym.ID, yytext()); }


//123.0 123.5e+10
[+-]?{cifra}+\.{cifra}+([eE][+-]{cifra}+)?  { return new Symbol(sym.DOUBLE_CONST, Double.parseDouble(yytext())); } 

//+341, -123, 123
[+-]?{cifra}+                        { return new Symbol(sym.INT_CONST, Integer.parseInt(yytext())); }

\"[^"]*\"       { return new Symbol(sym.STRING_CONST, yytext().substring(1, yytext().length - 1)); }

. { print("Greska" + yytext()); }
