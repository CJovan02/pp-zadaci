class MObject {
    public ArrayList<PairInfo> members;
}

class PairInfo {
    public string name;
    public Type type;
    public MObject? mObject;
    // Ovo stavljamo da samo za smenu "Pair -> ID : ID"
    // Da bi kasnije vrlo lako mogli da proverimo da li PairInfo drzi neki drugi ID.
    public string? anotherIdName;

    public PairInfo(string name, Type type, MObject? mObject = null, string? anotherIdName = null) {
        this.name = name;
        this.type = type;
        this.mObject = mObject;
        this.anotherIdName = anotherIdName;
    }
}
